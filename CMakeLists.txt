cmake_minimum_required(VERSION 3.20)
project(OverworldEvents LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 is required")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MSVC)
    message(FATAL_ERROR "Use clang-cl in an x64 Visual Studio tools environment")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
add_library(OverworldEvents SHARED src/OverworldEvents.cpp src/StateStore.cpp src/TravelLocations.cpp
            src/MapMarkers.cpp)
target_include_directories(OverworldEvents PRIVATE vendor/cwmods)
target_link_libraries(OverworldEvents PRIVATE user32)
include(CTest)

    add_test(NAME StateStore COMMAND StateStoreTest)
